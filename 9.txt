def max_of_three(a, b, c):
    if a > b and a > c:
        return a
    if b > c and b > c:
        return b
    if c > a and c > b:
        return c
    if a == b and b == c and a == c:
        return a
    if a == b and c > a or b:
        return c
    if a == c and b > a or c:
        return b
print(max_of_three(1, 2, 3))
print(max_of_three(10, 2, 3))
print(max_of_three(1, 1, 1))
print(max_of_three(-5, -2, -9))
